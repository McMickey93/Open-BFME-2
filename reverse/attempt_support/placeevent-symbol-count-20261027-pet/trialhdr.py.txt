#!/usr/bin/env python3
"""trialhdr.py IN OUT VARIANT : write a trial GameLogic header.
A: +0x9D field only (+2 symbols)
B: A + m_99/m_9b/m_9f/m_A0 + m_118/m_11C + m_6E (+9 symbols)
C: B + m_1B4 (+10 symbols)
D: A + m_A0 (+4 symbols)"""
import sys
src = open(sys.argv[1]).read()
v = sys.argv[3]
def rep(old, new):
    global src
    assert src.count(old) == 1, old
    src = src.replace(old, new)
if v == "A":
    rep("\tchar pad9B[0xAC - 0x9B];\n", "\tchar pad9B[0x9D - 0x9B];\n\tbool m_9d; // +0x9D\n\tchar pad9E[0xAC - 0x9E];\n")
if v == "D":
    rep("\tchar pad9B[0xAC - 0x9B];\n", "\tchar pad9B[0x9D - 0x9B];\n\tbool m_9d; // +0x9D\n\tchar pad9E[0xA0 - 0x9E];\n\tint m_A0; // +0xA0\n\tchar padA4[0xAC - 0xA4];\n")
if v in ("B", "C"):
    rep("\tchar pad99[0x9A - 0x99];\n", "\tbool m_99; // +0x99\n")
    rep("\tchar pad9B[0xAC - 0x9B];\n", "\tbool m_9b; // +0x9B\n\tchar pad9C[0x9D - 0x9C];\n\tbool m_9d; // +0x9D\n\tchar pad9E[0x9F - 0x9E];\n\tbool m_9f; // +0x9F\n\tint m_A0; // +0xA0\n\tchar padA4[0xAC - 0xA4];\n")
    rep("\tchar pad118[0x125 - 0x118];\n", "\tint m_118; // +0x118\n\tint m_11C; // +0x11C\n\tchar pad120[0x125 - 0x120];\n")
    rep("\tchar pad6E[0x98 - 0x6E];\n", "\tbool m_6E; // +0x6E\n\tchar pad6F[0x98 - 0x6F];\n")
if v == "C":
    rep("\tchar pad184[0x1B8 - 0x184];\n", "\tchar pad184[0x1B4 - 0x184];\n\tint m_1B4; // +0x1B4\n")
open(sys.argv[2], "w").write(src)
