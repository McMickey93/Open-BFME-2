#!/usr/bin/env python3
"""multi.py BASE OUT.cpp anchorkey=count ...  insert block-scope extern dummies before anchors."""
import sys
from pathlib import Path
ANCH = {
 "m1": 'TheInGameUI->message("GUI:NotEnoughMoneyToBuild");',
 "m2": 'TheInGameUI->message("GUI:ProductionQueueFull");',
 "m3": 'TheInGameUI->message("GUI:ParkingPlacesFull");',
 "m4": 'TheInGameUI->message("GUI:UnitMaxedOut");',
 "lbc": "int lbc = -1;",
 "snd": "static BfmeAudioEventPrefix136 noCanDoSound",
 "aud": "TheAudio->addAudioEvent(&noCanDoSound);",
 "start": "GameMessageDisposition disp = KEEP_MESSAGE;",
 "loc": "int localIndex = TheGhost",
 "top": "#include <list>",
 "aftgl": "#include \"Common/BfmeAudioEventPrefix136.h\"",
 "aftview": "struct PlacePixel : ICoord2D",
 "mouse": "class Mouse",
 "ingame": "class InGameUI",
 "fn": "GameMessageDisposition PlaceEventTranslator::translateGameMessage(const GameMessage *msg)",
 "pblk": "bool portBlocked = false;",
 "nw": "PlaceCoord newWorld;",
 "val": "int value = 0xFFFFFE;",
 "lst": "DrawableList list;",
 "inf": "PickAndPlayInfo info;",
}
lines = Path(sys.argv[1]).read_text().split("\n")
inserts = {}
for spec in sys.argv[3:]:
    k, c = spec.split("=")
    inserts[k] = int(c)
out = []
used = 0
for l in lines:
    for k, c in inserts.items():
        if ANCH[k] in l:
            ind = l[:len(l) - len(l.lstrip())]
            out += [f"{ind}extern int pd_{k}_{j};" for j in range(c)]
            used += 1
    out.append(l)
assert used == len(inserts), (used, inserts)
Path(sys.argv[2]).write_text("\n".join(out))
