#!/usr/bin/env python3
"""Compile a scratch copy of a source (under build/scratch_pet/) and compare
one function's bytes against retail, reporting differences with disassembly.

Usage: cmp.py <repo-root> <scratch.cpp> <symbol-substr> <rva> <size> [--dis]
"""
import subprocess
import sys
from pathlib import Path

root = Path(sys.argv[1]).resolve()
sys.path.insert(0, str(root / "tools"))
import build  # noqa: E402
import capstone  # noqa: E402

src = Path(sys.argv[2]).resolve()
wanted = sys.argv[3]
rva = int(sys.argv[4], 16)
size = int(sys.argv[5])
dis = "--dis" in sys.argv

out = root / "build" / "scratch_pet" / (src.stem + ".obj")
out.parent.mkdir(parents=True, exist_ok=True)
if out.exists():
    out.unlink()
command, env = build.compiler_command(src, out)
proc = subprocess.run(command, cwd=root, env=env, capture_output=True, text=True)
if proc.returncode != 0:
    sys.stdout.write(proc.stdout or "")
    sys.stderr.write(proc.stderr or "")
    raise SystemExit("compile failed")
defined = build.defined_code_symbols(out)
matches = [s for s in defined if s.startswith("?") and wanted in s]
if len(matches) != 1:
    raise SystemExit(f"matches: {matches}")
compiled, relocs = build.read_object_symbol_bytes(out, matches[0], code_only=True)
retail = build.read_target_bytes(rva, size)
masked = bytearray(compiled[:size])
relocated = set()
for off, rtype, name in relocs:
    if off < size:
        for i in range(4):
            relocated.add(off + i)
diffs = [i for i in range(min(size, len(compiled))) if i not in relocated and compiled[i] != retail[i]]
print(f"{len(compiled)}B vs {size}B; {len(diffs)} non-reloc differing bytes"
      + (f", first +0x{diffs[0]:X}" if diffs else ""))
if dis or diffs:
    md = capstone.Cs(capstone.CS_ARCH_X86, capstone.CS_MODE_32)
    a = list(md.disasm(bytes(compiled[:size]), 0))
    b = list(md.disasm(bytes(retail), 0))
    dset = set(diffs)
    bi = {ins.address: ins for ins in b}
    for ins in a:
        bad = any((ins.address + k) in dset for k in range(ins.size))
        if bad or dis:
            r = bi.get(ins.address)
            rt = f"{r.mnemonic} {r.op_str}" if r else "?"
            print(f"{'*' if bad else ' '} {ins.address:04X}: {ins.mnemonic} {ins.op_str:40s} | {rt}")
